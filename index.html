Código de Google Apps Script: 

function doPost(e) {
  try {
    var raw = e.postData.contents;
    var data = JSON.parse(raw);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. CLIENTES_Y_SALDOS_POR_MARCA
    var sheetClientes = getOrCreateSheet(ss, "CLIENTES_SALDOS_MARCAS");
    sheetClientes.clear();
    sheetClientes.appendRow([
      "ID", "Cédula/RIF", "Nombre", "Teléfono", "Dirección",
      "Deuda Boutique ($)", "Deuda Exclusive ($)", "Deuda Beauty ($)",
      "Total Compras ($)", "Total Abonado ($)", "Saldo Deudor Total ($)", "Estado"
    ]);
    
    (data.clients || []).forEach(function(c) {
      sheetClientes.appendRow([
        c.id, c.doc, c.name, c.phone, c.address || '',
        c.debtBoutique || 0, c.debtExclusive || 0, c.debtBeauty || 0,
        c.totalPurchases, c.totalPaid, c.totalBalance,
        (c.totalBalance > 0 ? "DEUDOR" : "AL DÍA")
      ]);
    });

    // 2. HISTORIAL_COMPRAS
    var sheetCompras = getOrCreateSheet(ss, "HISTORIAL_COMPRAS");
    sheetCompras.clear();
    sheetCompras.appendRow(["Fecha", "Marca", "Cliente", "Cédula", "Artículos", "Monto USD", "Notas"]);
    (data.purchases || []).forEach(function(p) {
      sheetCompras.appendRow([p.date, p.brand, p.clientName, p.clientDoc, p.concept, p.amount, p.notes]);
    });

    // 3. HISTORIAL_ABONOS
    var sheetAbonos = getOrCreateSheet(ss, "HISTORIAL_ABONOS");
    sheetAbonos.clear();
    sheetAbonos.appendRow(["Fecha", "Marca Abonada", "Cliente", "Cédula", "Método", "Referencia", "Monto USD", "Notas"]);
    (data.payments || []).forEach(function(a) {
      sheetAbonos.appendRow([a.date, a.brand, a.clientName, a.clientDoc, a.method, a.ref, a.amount, a.notes]);
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Respaldo actualizado correctamente en Google Sheets",
      updatedAt: new Date().toLocaleString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  return sheet;
}
